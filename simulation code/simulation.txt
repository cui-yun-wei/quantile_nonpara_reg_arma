
#####################################################################################################
# Step 1:
# Install the R package reticulate, configure a Python environment, and install
# the required Python packages (NumPy and SciPy).
#
# Python package purposes:
#   NumPy   Used for numerical array operations and data storage/transmission.
#   SciPy   Used for numerical optimization routines.
#
# Tested Python package versions:
#   NumPy   1.26.4
#   SciPy   1.15.3
#####################################################################################################

# Install and load the reticulate package.
install.packages("reticulate")
library(reticulate)

# Install Miniconda (required only for the first setup).
install_miniconda()

# Create a Conda environment named "r-reticulate".
conda_create("r-reticulate")

# Verify that the environment "r-reticulate" has been created successfully.
conda_list()

# Install the required Python packages in the "r-reticulate" environment.
py_install(
  c("numpy", "scipy"),
  envname = "r-reticulate"
)

# Activate the "r-reticulate" environment and verify that the required
# Python packages are available.
use_condaenv("r-reticulate", required = TRUE)

py_module_available("numpy")
py_module_available("scipy")

#####################################################################################################
# Step 2:
# Install the required R packages.
#
# Package purposes:
#   nimble      Used for simulating the double exponential distribution.
#   splines     Used for generating spline basis functions.
#   data.table  Used for efficiently writing simulation results to .txt files.
#
# Tested package versions:
#   nimble      1.3.0
#   splines     4.3.2
#   data.table  1.17.0
#####################################################################################################

# Install the R packages required for the simulation.
install.packages(c("nimble", "splines", "data.table"))



#####################################################################################################
# Step 3:
# Download the .py and .R files to the current working directory and run the
# simulation. The simulation results will be saved as .txt files in the same
# working directory.
#
# Function arguments:
#   alpha1         AR(2) coefficient for lag 1 (default: 0.5).
#   alpha2         AR(2) coefficient for lag 2 (default: 0.1).
#   sim_len        Number of simulation repetitions (default: 1000).
#   tau            Quantile level (e.g., 0.25, 0.50, or 0.75).
#   innovation_dis Innovation distribution. Available options are:
#                    "double_exponential"
#                    "t"
#                    "normal"
#                    "chisq5"
#                    "chisq15"
#                  Default: "t".
#   J_n            Number of interior knots for spline basis functions
#                  (default: 9).
#
# For each combination of alpha1, alpha2, sim_len, tau, and innovation_dis,
# a total of 9 .txt files will be generated. These files contain simulation
# results for the smooth functions F1, F2, and F3 on [0,1], with three different sample
# path lengths: 200, 400, and 1000.
#
# if (fct=='F1') fx_t = 1 - 6*x_t + 36*x_t^2 - 53*x_t^3 + 22*x_t^5
# if (fct=='F2') fx_t = sin(x_t*2*pi) + 2*x_t^2
# if (fct=='F3') fx_t = atan((x_t-0.5)*5) - x_t^2/3
#
#####################################################################################################

# Load the R function for simulation.
source("regression_timeseries_quantile_ar2_simu.R")

# Run the simulation with specified parameter settings.
regression_timeseries_quantile_ar2_simu(
    alpha1 = 0.5,
    alpha2 = 0.1,
    sim_len = 1000,
    tau = 0.2,
    innovation_dis = "chisq5",
    J_n = 9
)
